Новая таблица — `merchant_settlements`:

```text
-- Planned: migration 044-merchant-settlements.sql
CREATE TABLE merchant_settlements (
  id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  instance_id           UUID NOT NULL REFERENCES instances(id),
  merchant_id           UUID NOT NULL REFERENCES merchants(id),
  settlement_number     TEXT NOT NULL,           -- например, STL-2026-03-ACME-001
  period_start          DATE NOT NULL,
  period_end            DATE NOT NULL,
  cheques_activated     INT NOT NULL DEFAULT 0,
  cheques_returned      INT NOT NULL DEFAULT 0,
  face_value_usdc       NUMERIC(18,6) NOT NULL,  -- общий номинал активированных чеков
  settlement_currency   TEXT NOT NULL DEFAULT 'USD',
  fx_rate               NUMERIC(12,6) NOT NULL DEFAULT 1.0,  -- USDC/settlement_currency
  fx_rate_source        TEXT DEFAULT 'manual',   -- 'manual', 'ecb', 'coinbase'
  amount_due            NUMERIC(18,6) NOT NULL,  -- face_value_usdc × fx_rate
  status                TEXT NOT NULL DEFAULT 'draft'
                          CHECK (status IN ('draft','issued','paid','overdue','void')),
  issued_at             TIMESTAMPTZ,
  due_at                TIMESTAMPTZ,
  paid_at               TIMESTAMPTZ,
  payment_ref           TEXT,                    -- wire ref, Solana tx sig, etc.
  notes                 TEXT,
  created_at            TIMESTAMPTZ DEFAULT NOW(),
  updated_at            TIMESTAMPTZ DEFAULT NOW()
);

CREATE UNIQUE INDEX settlements_merchant_period
  ON merchant_settlements(merchant_id, period_start, period_end);
```

Новая таблица — `merchant_rebates`:

```text
-- Запланировано: migration 044-merchant-settlements.sql, та же миграция
CREATE TABLE merchant_rebates (
  id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  instance_id           UUID NOT NULL REFERENCES instances(id),
  merchant_id           UUID NOT NULL REFERENCES merchants(id),
  settlement_id         UUID NOT NULL REFERENCES merchant_settlements(id),
  rebate_number         TEXT NOT NULL,           -- например, RBT-2026-03-ACME-001
  settled_volume_usdc   NUMERIC(18,6) NOT NULL,
  rebate_pct_applied    NUMERIC(5,3) NOT NULL,
  rebate_amount_usdc    NUMERIC(18,6) NOT NULL,
  tier_used             JSONB,                   -- снимок того tier, который сработал
  status                TEXT NOT NULL DEFAULT 'pending'
                          CHECK (status IN ('pending','paid','void')),
  scheduled_at          TIMESTAMPTZ,             -- settlement.paid_at + delay_days
  paid_at               TIMESTAMPTZ,
  tx_hash               TEXT,                    -- Solana tx, если выплата была on-chain
  created_at            TIMESTAMPTZ DEFAULT NOW()
);
```
